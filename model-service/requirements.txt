# The model's own runtime, pinned exactly as the image that writes the
# model pins it (docker/requirements-training.txt, numpy and pandas
# through src/requirements.txt). A joblib artifact is a pickle of
# scikit-learn's own objects, so the version that loads it has to match
# the version that wrote it -- a serving image drifting from the training
# image is a textbook way to get an InconsistentVersionWarning and,
# eventually, a model that either fails to unpickle or silently
# mispredicts. The training side used to be unpinned under a "bump both
# together" note; tests/test_requirements.py now fails if the two differ.
scikit-learn==1.9.1
numpy==2.5.3
pandas==3.0.5
joblib==1.6.0
